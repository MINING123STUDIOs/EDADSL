\chapter{Formal Grammar}
The Formal Grammar of EDADSL is defined using Extended Backus–Naur form, while Primitives are given using Regular Expressions.
\section{Primitives}
The Primitives are defined using Regular Expressions. For more Information on Lexical structure see Part I Chapter 3.
\begin{verbatim}
idchar  = [_a-zA-Z0-9]
id      = [_a-zA-Z0-9]+
letter  = [a-zA-Z]
\end{verbatim}

\section{General Grammar}
\begin{verbatim}
program                       = "start", { statement, 
                                { ("%N" | newline) } }

statement                     = assignment
                              | flow_control
                              | function
                              | electrical_system_statement
                              | compound_assignment
                              | postfix_expression

assignment                    = variable_assignment          
                              | constant_assignment

function                      = function_call_built_in
                              | function_call_ud
                              | function_definition

flow_control                  = conditional
                              | iterator
                              | jump
                              | halt
                              | noop

conditional                   = "if", boolean, "::", code_block,
                              ["else", "::", code_block]

iterator                      = "for", variable, "E",
                                (list | set), "::", code_block

jump                          = ( "gotomarker" | "gotojump" ), id

function_definition = "function", "f.u.", id, 
                      "(", [ { id, "," }, id], 
                      ")", "::", "{", f_statement, 
                      {f_statement}, "}"

f_statement = statement | "return", [data]

function_call_ud              = "f.u.", id, "(",
                                [{ data, "," }, data] ,")"

function_call_built_in        = "f.", id, "(",
                                [{ data, "," }, data] ,")"

variable_assignment           = variable,
                                (typed_assign | inferred_assign),
                                [";", "M", "=", boolean]

typed_assign                  = assigner
inferred_assign               = "t", "=", data

assigner                      = "[bool]=",   boolean_type
                              | "[int]=",     int_type
                              | "[real]=",    real_type
                              | "[list]=",    list_type
                              | "[string]=",  string_type
                              | "[set]=",     set_type
                              | electrical_system_assigner

constant_assignment = "let", "c.u.", id, assigner, 
                      [";", "P", "=", boolean]


halt                          = "halt"
noop                          = "skip"

list                          = "[", [{ data, "," }, data], "]"

variable                      = "$", id

code_block                    = "{", statement, 
                                { ("%N" | newline), statement }, "}"

data                          = variable
                              | literal
                              | constant
                              | function_call_built_in
                              | function_call_ud
                              | expression
\end{verbatim}

\section{Grammar relating to the electrical System}
\begin{verbatim}
electrical_system_statement   = make_statement
                              | elec_statement
                              | phys_statement
                              | syst_statement

electrical_system_assigner    = "[part]=", part_type
                              | "[connection]=", connection_type
                              | "[pin]=", pin_type
                              | "[net]=", net_type
\end{verbatim}

\subsection{Make Statement}
\begin{verbatim}
make_statement = "make.", ("pcb" | "schematic")
\end{verbatim}

\subsection{Electrical Statements}
\begin{verbatim}
elec_statement = "elec.",
  ( "part", part_reference, ":", model_name,
    "(", [model_params], ")",
    [";", "des", "=", string_type],
    [";", "fp", "=", string_type]
  | "connect", (net_type | pin_type),
    pin_type, { pin_type }
  | "net", string_type, { string_type }
  | "tag", net_type, string_type,
    { string_type }, ["-remove"]
  | "untag", net_type, string_type,
    { string_type }
  )

part_reference = string_type
               | "[", { string_type, "," }, string_type, "]"

model_name     = id
model_params   = [{ id, "=", data, "," }, id, "=", data]
\end{verbatim}

\subsection{Physical Constraints}
\begin{verbatim}
phys_statement = phys_prox | phys_creep | phys_layer
               | phys_side | phys_board | phys_shrinknet
               | phys_mkplane | phys_viafill
               | phys_connectioncurrent | phys_align
               | phys_connmatch | phys_mktext

phys_prox   = "phys.prox", set,
              ( set, "min", real, "max", real
              | "minim" ), priority

phys_creep  = "phys.creep", set, set,
              "d", real, priority

phys_layer  = "phys.layer", set, layer_ref, priority

phys_side   = "phys.side", set, layer_ref, priority

phys_board  = "phys.board",
              "N=", int,
              "T=", list,
              "W=", list,
              ["M=", list]

phys_shrinknet = "phys.shrinknet", net, priority

phys_mkplane   = "phys.mkplane", layer_ref, net

phys_viafill   = "phys.viafill", pad,
              "v", real, "d", real,
              ["p", pattern]

phys_connectioncurrent = "phys.connectioncurrent",
              (connection | set), real, priority

phys_align  = "phys.align", list,
              "axis", ("x" | "y"),
              "d", real, priority

phys_connmatch = "phys.connmatch",
              match_property, set, priority

phys_mktext = "phys.mktext",
              "txt", string,
              ["font", string],
              ["size", real],
              ["material", string],
              ["layer", layer_ref]
\end{verbatim}

\subsection{System Statements}
\begin{verbatim}
syst_statement = syst_flag | syst_cm | syst_alg

syst_flag = "syst.flag.", flag_name, real

syst_cm   = "syst.cm.calc",
            ("approx" | "calculus" | "hybrid")

syst_alg  = "syst.alg", id
\end{verbatim}

\subsection{Common Rules}
\begin{verbatim}
layer_ref      = int | "T" | "B"
priority       = ["-d" | "-nth" | "-imp" | "-vimp"
                | "-oeo" | "-eo" | "-ne"]
pattern        = "grid" | "hex" | "stagger"
               | "radial" | "random"
match_property = "l" | "res" | "L" | "cap"
               | "imp" | "t"
flag_name      = "vias" | "slots" | "jumpers"
               | "solder_reinforced_tracks"
               | "buried_vias"
               | "approx_eps" | "approx_eps_mult"
\end{verbatim}
