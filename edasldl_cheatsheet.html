Compound Assignment

# Arithmetic (forward & reverse)
$x += 3    $x -= 3    $x *= 3    $x /= 3
$x %= 3    $x ^= 3
$x +r= 3   #c# = 3 + $x

# Bitwise / shift
$x <<= 2   $x >>= 2
$x <<<[32]= 1  $x >>>[32]= 1

# Boolean / bitwise logic
$b &&= expr   $b ||= expr   $b ^^= expr
$b b&&= expr  $b b||= expr  $b b^^= expr

# Tensor in-place
$m ~.T       #c# transpose in place
$m ~.H       #c# Hermitian transpose
$m .*= B     #c# matrix multiply

Control Flow

# if / otherwise / else
if $x > 0 :: {
    f.util.print("positive")
} otherwise $x =n= 0 :: {
    f.util.print("zero")
} else :: {
    f.util.print("negative")
}

# for loop
for $i E [1,2,3] :: {
    f.util.print($i)
}

# while
while $running :: {
    # ...
}

# do-while
do :: {
    $i += 1
} while $i < 10

# break, continue, halt, skip
break     continue     halt     skip

Physical Constraints

# Board definition
phys.board
    N=2
    T=[0.15, 1.6, 0.15]   #c# thickness [m]
    W=[0.1, 0.1]        #c# width [m]
    M=[1, 0]             #c# layers (optional)

# Proximity
phys.prox {R1, C1}
    min=1m max=50m -d

# Creepage

    d=5m -imp

# Layer assignment
phys.layer {R1, R2} B -vimp

Scope Rules

  • Block scoping everywhere
  • Top-level variables: global scope
  • Block-local variables destroyed on exit
  • Variables must be declared before use
  • Redeclaration allowed (new type overwrites)
  • Lambdas: read-only closure, local vars OK
  • Functions: local scope
